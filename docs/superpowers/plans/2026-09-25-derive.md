# `tl-ds derive` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `node tools/derive.mjs acme.primitives.json` schreibt aus einer Kunden-JSON eine Theme-Datei `acme.theme.css`, die nur abweichende Rollen unter `[data-tl-theme="acme"]` trägt, gerechnete Werte WCAG-fest hält und gelieferte Werte mit Befund durchlässt.

**Architecture:** Der Kunde liefert Primitive (Saat oder Stufen). `derive` legt gelieferte Stufen über `src/primitives.json`, löst `src/tokens.json` damit auf (bestehendes `resolve`), überlagert die aus Saat gerechneten Rollen (Marke, Kategorien, Beschriftungen), vergleicht mit dem TopLogic-Ergebnis und schreibt nur die Differenz als Rollen-CSS. Eine geteilte Paarliste (`tools/lib/pairs.mjs`) prüft Ergebnis und Paket mit denselben Regeln.

**Tech Stack:** Node ≥ 22, ESM, `node --test`, keine Abhängigkeiten. Alles unter `design-system/`.

**Spec:** `docs/superpowers/specs/2026-09-25-derive-design.md`

```js
/** Schrittweite in L, mit der aufKontrast() einen Wert nachführt. */
export const KONTRAST_SCHRITT = 0.005;

/**
 * Führt einen Wert in L nach, bis er gegen jeden Grund das geforderte Verhältnis erreicht.
 * Farbton und Chroma bleiben; das ist dieselbe Regel wie bei oklchZuHex(): L und H sind die
 * Aussage, C die Stellschraube — hier ist L die Stellschraube, weil Kontrast Helligkeit ist.
 * Ein Wert, der schon trägt, kommt unverändert zurück (angepasst: false).
 */
export function aufKontrast(hex, gruende, soll, richtung) {
  const { L, C, H } = hexZuOklch(hex);
  const traegt = (h) => gruende.every((g) => ratio(h, g) >= soll);
  if (traegt(hex)) return { hex: hex.toUpperCase(), angepasst: false, L };
  const schritt = richtung === 'dunkler' ? -KONTRAST_SCHRITT : KONTRAST_SCHRITT;
  for (let l = L + schritt; l > 0 && l < 1; l += schritt) {
    const { hex: probe } = oklchZuHex({ L: l, C, H });
    if (traegt(probe)) return { hex: probe, angepasst: true, L: l };
  }
  throw new Error(`${hex} erreicht ${soll} gegen ${gruende.join(', ')} in keiner Helligkeit`);
}
```

```js
const GRUENDE_TEXT = ['surface-layer', 'surface-layer-nested', 'surface-overlay',
  'surface-interactive-hover', 'surface-interactive-selected', 'surface-interactive-active'];
const GRUENDE_FOKUS = ['surface-base', 'surface-layer', 'surface-layer-nested', 'surface-overlay',
  'surface-field', 'surface-brand-subtle'];
const STATUS = ['error', 'warning', 'success', 'info'];
const KATEGORIEN = [1, 2, 3, 4, 5, 6, 7, 8];

export function kontrastPaare(T) {
  const p = [];
  const paar = (fg, bg, soll) => p.push({ fg, bg, soll });
  paar('text-primary', 'surface-base', 16.0); paar('text-secondary', 'surface-base', 7.0);
  paar('text-helper', 'surface-base', 5.2); paar('text-placeholder', 'surface-field', 4.6);
  paar('text-link', 'surface-base', 4.5); paar('text-link-hover', 'surface-base', 4.5);
  paar('icon-primary', 'surface-base', 3.0); paar('icon-secondary', 'surface-base', 3.0);
  paar('border-emphasis', 'surface-base', 4.6); paar('border-control', 'surface-base', 3.0);
  paar('border-interactive', 'surface-base', 3.0); paar('focus-ring', 'surface-base', 3.0);
  paar('text-link', 'surface-brand-subtle', 4.5); paar('text-primary', 'surface-brand-subtle', 4.5);
  for (const bg of GRUENDE_TEXT) paar('text-primary', bg, 4.5);
  for (const s of STATUS) {
    for (const bg of ['surface-base', 'surface-layer']) {
      paar(`status-${s}-text`, bg, 4.5); paar(`status-${s}-border`, bg, 3.0);
    }
    paar('text-primary', `status-${s}-subtle`, 4.5);
  }
  for (const bg of GRUENDE_FOKUS) paar('focus-ring', bg, 3.0);
  for (const k of KATEGORIEN) {
```

```js
/** Chroma eines temperierten Graus — so viel, wie unser heutiges kühles Grau trägt. */
export const GRAU_CHROMA = 0.008;
/** Farbton je Temperatur in Grad: warm ein gelbliches, cool ein bläuliches Grau. */
export const GRAU_TON = { warm: 70, cool: 250 };

/**
 * Eine Graurampe auf den Helligkeiten der TopLogic-Rampe. Nur der Farbton wechselt, nie L —
 * deshalb hält jedes Kontrastpaar, das mit unserem Grau hält, auch mit diesem.
 */
export function grauRampe(temperatur, stufen) {
  if (!(temperatur === 'neutral' || temperatur in GRAU_TON)) throw new Error(`Grau-Temperatur unbekannt: ${temperatur}`);
  const C = temperatur === 'neutral' ? 0 : GRAU_CHROMA;
  const H = temperatur === 'neutral' ? 0 : (GRAU_TON[temperatur] * Math.PI) / 180;
  return Object.fromEntries(Object.entries(stufen).map(([n, hex]) => [n, oklchZuHex({ L: hexZuOklch(hex).L, C, H }).hex]));
}
```

```js
const grad = (rad) => ((rad * 180) / Math.PI + 360) % 360;
const rad = (g) => (g * Math.PI) / 180;

/**
 * Verteilt `anzahl` Farbtöne auf dem Kreis so, dass jeder mindestens `mindest` Grad zu jedem
 * blockierten Ton und zu jedem anderen hält. Verfahren: die Lücken zwischen den blockierten
 * Tönen bekommen reihum den nächsten Punkt, immer die Lücke, in der der Abstand danach am
 * größten bleibt. Reicht der Kreis nicht, sinkt der Mindestabstand in Schritten von 2°.
 */
export function kategorienVerteilen(blockiert, anzahl = 8, mindest = 20) {
  const feste = [...new Set(blockiert.map((g) => ((g % 360) + 360) % 360))].sort((a, b) => a - b);
  for (let abstand = mindest; abstand > 0; abstand -= 2) {
    const luecken = feste.map((von, i) => {
      const bis = i + 1 < feste.length ? feste[i + 1] : feste[0] + 360;
      return { von, breite: bis - von, punkte: 0 };
    });
    for (let n = 0; n < anzahl; n++) {
      let beste = null;
      for (const l of luecken) {
        const spanne = l.breite / (l.punkte + 2);
        if (spanne >= abstand && (!beste || spanne > beste.breite / (beste.punkte + 2))) beste = l;
      }
      if (!beste) break;
      beste.punkte++;
    }
    const gesetzt = luecken.reduce((s, l) => s + l.punkte, 0);
    if (gesetzt < anzahl) continue;
    const toene = luecken.flatMap((l) => Array.from({ length: l.punkte },
      (_, k) => Math.round((l.von + (l.breite * (k + 1)) / (l.punkte + 1)) % 360)));
    return { toene, abstand };
  }
  throw new Error(`${anzahl} Kategorien finden keinen Platz`);
}

export const KATEGORIE_SUBTLE = { light: { L: 0.955, C: 0.03 }, dark: { L: 0.27, C: 0.05 } };
export const KATEGORIE_BORDER_CHROMA = 0.13;

/** Eine Kategorie aus einem Farbton: getönte Fläche und Linie je Modus, Linie der hellste
 *  (hell) bzw. dunkelste (dunkel) Wert mit 3,0 auf beiden Gründen. */
export function kategorieRollen(tonGrad, gruende) {
  const H = rad(tonGrad);
  const subtle = (m) => oklchZuHex({ ...KATEGORIE_SUBTLE[m], H }).hex;
  const border = (m) => aufKontrast(oklchZuHex({ L: m === 'light' ? 0.75 : 0.45, C: KATEGORIE_BORDER_CHROMA, H }).hex,
    gruende[m], 3.0, m === 'light' ? 'dunkler' : 'heller').hex;
  return {
    light: { subtle: subtle('light'), border: border('light') },
    dark: { subtle: subtle('dark'), border: border('dark') },
  };
}
```

Hinweis: `grad` und `rad` sind Modul-lokal; `farbtonAbstand` darf `grad` mitbenutzen, aber nicht umbauen.

```js
// Liest die Kundendatei der Stufe A und prüft sie gegen das Paket. Ein Fehler nennt den Pfad in
// der JSON; nichts wird geraten.
const HEX = /^#[0-9a-fA-F]{6}$/;
const THEME = /^[a-z][a-z0-9-]*$/;
const FAMILIEN = ['color', 'font', 'measure', 'motion', 'effect'];
const BEDEUTUNGEN = ['red', 'amber', 'green', 'teal'];
const TEMPERATUREN = ['warm', 'neutral', 'cool'];
const KATEGORIEN = Array.from({ length: 8 }, (_, i) => `category-${i + 1}`);

export class ThemeEingabeFehler extends Error {
  constructor(pfad, meldung) { super(`${pfad}: ${meldung}`); this.pfad = pfad; }
}

const hex = (pfad, v) => {
  if (typeof v !== 'string' || !HEX.test(v)) throw new ThemeEingabeFehler(pfad, `kein Hexwert #rrggbb: ${JSON.stringify(v)}`);
  return v.toUpperCase();
};
const objekt = (pfad, v) => {
  if (!v || typeof v !== 'object' || Array.isArray(v)) throw new ThemeEingabeFehler(pfad, 'ein Objekt erwartet');
  return v;
};

export function leseThemeEingabe(json, P, T) {
  const wurzel = objekt('', json);
  if (typeof wurzel.theme !== 'string' || !THEME.test(wurzel.theme) || wurzel.theme === 'tl')
    throw new ThemeEingabeFehler('theme', 'Pflicht, [a-z][a-z0-9-]*, nie "tl"');
  for (const k of Object.keys(wurzel)) if (k !== 'theme' && !FAMILIEN.includes(k))
    throw new ThemeEingabeFehler(k, `unbekannte Familie, erlaubt: ${FAMILIEN.join(', ')}`);

  const geliefert = {};
  const saat = {};
  const markenFamilie = /^\{([a-z]+)-\d+\}$/.exec(T.color.light['surface-brand'])[1];

  const stufen = (pfad, familie, v) => {
    for (const [stufe, wert] of Object.entries(objekt(pfad, v))) {
      const name = `${familie}-${stufe}`;
      if (!(name in P.color)) throw new ThemeEingabeFehler(`${pfad}.${stufe}`, `Stufe ${name} gibt es nicht`);
      geliefert[name] = hex(`${pfad}.${stufe}`, wert);
    }
  };

  if (wurzel.color !== undefined) {
    const c = objekt('color', wurzel.color);
    for (const k of Object.keys(c)) if (!['brand', 'gray', 'category', ...BEDEUTUNGEN].includes(k))
      throw new ThemeEingabeFehler(`color.${k}`, 'unbekannt; erlaubt: brand, gray, category, red, amber, green, teal');
    if (typeof c.brand === 'string') saat.brand = hex('color.brand', c.brand);
    else if (c.brand !== undefined) {
      stufen('color.brand', markenFamilie, c.brand);
      if (!(`${markenFamilie}-49` in geliefert)) throw new ThemeEingabeFehler('color.brand.49', 'die Stufe 49 ist die Saat und fehlt');
      saat.brand = geliefert[`${markenFamilie}-49`];
    }
    if (typeof c.gray === 'string') {
      if (!TEMPERATUREN.includes(c.gray)) throw new ThemeEingabeFehler('color.gray', `warm, neutral oder cool, nicht ${JSON.stringify(c.gray)}`);
      saat.gray = c.gray;
    } else if (c.gray !== undefined) stufen('color.gray', 'gray', c.gray);
    if (c.category !== undefined) {
      saat.category = {};
      for (const [k, v] of Object.entries(objekt('color.category', c.category))) {
        if (!KATEGORIEN.includes(k)) throw new ThemeEingabeFehler(`color.category.${k}`, 'category-1 bis category-8');
        saat.category[k] = hex(`color.category.${k}`, v);
      }
    }
    for (const b of BEDEUTUNGEN) if (c[b] !== undefined) stufen(`color.${b}`, b, c[b]);
  }
  for (const f of FAMILIEN.slice(1)) if (wurzel[f] !== undefined) {
    for (const [stufe, wert] of Object.entries(objekt(f, wurzel[f]))) {
      if (!(stufe in P[f])) throw new ThemeEingabeFehler(`${f}.${stufe}`, `Stufe gibt es nicht in ${f}`);
      if (typeof wert !== 'string' && typeof wert !== 'number') throw new ThemeEingabeFehler(`${f}.${stufe}`, 'ein Wert erwartet');
      geliefert[stufe] = String(wert);
    }
  }
  return { theme: wurzel.theme, geliefert, saat };
}
```

```js
// ist; schreibt nur Rollen, die von TopLogic abweichen; meldet, was ein Paar verfehlt.
// Spec: docs/superpowers/specs/2026-09-25-derive-design.md
import { resolve, primitives } from './resolve.mjs';
import { grauRampe, kategorienVerteilen, kategorieRollen, markenRollen, beschriftungAbleiten, hexZuOklch, aufKontrast }
  from './lib/color.mjs';
import { kontrastPaare, werte } from './lib/pairs.mjs';
import { leseThemeEingabe } from './lib/theme-input.mjs';
import { ratio } from './lib/contrast.mjs';

const REF = /^\{([a-z0-9-]+)\}$/;
const MODES = ['light', 'dark'];
const KATEGORIEN = Array.from({ length: 8 }, (_, i) => i + 1);
const grad = (hex) => ((hexZuOklch(hex).H * 180) / Math.PI + 360) % 360;

const HERKUNFT = (theme, quelle, anzahl) => `/* Generated by TopLogic/tl-design-system, tools/derive.mjs, from ${quelle}.
   Do not edit — edit the JSON and regenerate. Loaded after tokens.css; applies under data-tl-theme="${theme}".
   Findings: ${anzahl} ${anzahl === 1 ? 'pair' : 'pairs'} below its threshold${anzahl ? ' (see report at the end of this file)' : ''}. */\n`;

/** Primitive, die eine Rolle in einem Modus liest — für die Herkunft (geliefert oder gerechnet). */
function primitiveVon(T, mode, rolle) {
  const m = REF.exec(String(T.color[mode][rolle] ?? T.measure[rolle] ?? ''));
  return m ? m[1] : null;
}

export function derive(P, T, json, quelle = 'input') {
  const eingabe = leseThemeEingabe(json, P, T);
  const { theme, geliefert, saat } = eingabe;
  const hinweise = [];

  // 1 Gelieferte Stufen und die Graurampe liegen über den Primitiven.
  const Pk = structuredClone(P);
  if (saat.gray) {
    const grau = Object.fromEntries(Object.entries(P.color).filter(([n]) => n.startsWith('gray-')));
    Object.assign(Pk.color, grauRampe(saat.gray, grau));
  }
  for (const [n, v] of Object.entries(geliefert)) for (const f of Object.keys(Pk)) if (typeof Pk[f] === 'object' && n in Pk[f]) Pk[f][n] = v;

  // 2 Auflösen: geliefertes zieht durch alle Rollen, Beschriftungen folgen den Flächen.
  const R0 = resolve(P, T);
  const Rk = resolve(Pk, T);
  const gruende = Object.fromEntries(MODES.map((m) => [m, [Rk.color[m]['surface-base'], Rk.color[m]['surface-layer']]]));

  // 3 Gerechnete Rollen überlagern, wo keine gelieferte Primitive dahintersteht.
  const setze = (mode, rolle, hex) => {
    const prim = primitiveVon(T, mode, rolle);
    if (prim && prim in geliefert) return;
    Rk.color[mode][rolle] = hex;
  };
  if (saat.brand) {
    const m = markenRollen(saat.brand, gruende);
    hinweise.push(...m.hinweise);
    for (const mode of MODES) for (const [rolle, hex] of Object.entries(m[mode])) setze(mode, rolle, hex);
  }
  const markenTon = grad(Rk.color.light['surface-brand']);
  const bedeutungen = ['error', 'warning', 'success', 'info'].map((s) => grad(Rk.color.light[`status-${s}-surface`]));
  for (const b of bedeutungen) {
    const d = Math.abs(((markenTon - b) % 360 + 540) % 360 - 180);
    if (d < 20) hinweise.push(`die Marke liegt ${d.toFixed(0)}° neben einer Bedeutung; Stufe B entscheidet, ob die Bedeutung einen anderen Ton bekommt`);
  }
  const eigene = saat.category ?? {};
  const fehlende = KATEGORIEN.filter((k) => !(`category-${k}` in eigene));
  const { toene, abstand } = kategorienVerteilen([markenTon, ...bedeutungen, ...Object.values(eigene).map(grad)], fehlende.length);
  if (abstand < 20) hinweise.push(`die Kategorien halten nur ${abstand}° zueinander; der Kreis war mit dieser Marke voller als bei TopLogic`);
  const toneJeKategorie = Object.fromEntries([...KATEGORIEN.map((k) => [k, eigene[`category-${k}`] ? grad(eigene[`category-${k}`]) : null])]);
  fehlende.forEach((k, i) => { toneJeKategorie[k] = toene[i]; });
  for (const k of KATEGORIEN) {
    const r = kategorieRollen(toneJeKategorie[k], gruende);
    for (const mode of MODES) { setze(mode, `category-${k}-subtle`, r[mode].subtle); setze(mode, `category-${k}-border`, r[mode].border); }
  }

  // 4 Beschriftungen für jede Fläche neu, mit den Kandidaten des Kundengraus.
  const prim = primitives(Pk);
  const kandidaten = { dunklerKandidat: prim[REF.exec(T.derived.candidates.dark)[1]], hellerKandidat: prim[REF.exec(T.derived.candidates.light)[1]] };
  for (const mode of MODES) for (const [name, { surface }] of Object.entries(T.derived[mode]))
    Rk.derived[mode][name] = beschriftungAbleiten(Rk.color[mode][surface], kandidaten).hex;

  // 5 Differenz: beide Blöcke tragen dieselben Farbrollen.
  const farbRollen = [...Object.keys(T.color.light), ...Object.keys(T.derived.light)].filter((k) => !k.startsWith('_'));
  const effektRollen = Object.keys(T.effect.light).filter((k) => !k.startsWith('_'));
  const abweichend = (r) => MODES.some((m) => werte(Rk, m, r) !== werte(R0, m, r));
  const U = farbRollen.filter(abweichend);
  const E = effektRollen.filter((r) => MODES.some((m) => Rk.effect[m][r] !== R0.effect[m][r]));
  const rollen = { light: {}, dark: {}, measure: {} };
  for (const mode of MODES) {
    for (const r of U) rollen[mode][r] = werte(Rk, mode, r);
    for (const r of E) rollen[mode][r] = Rk.effect[mode][r];
  }
  for (const r of Object.keys(T.measure).filter((k) => !k.startsWith('_'))) if (Rk.measure[r] !== R0.measure[r]) rollen.measure[r] = Rk.measure[r];

  // 6 Befunde: nur gelieferte Werte dürfen ein Paar verfehlen.
  const befunde = [];
  for (const mode of MODES) for (const { fg, bg, soll } of kontrastPaare(T)) {
    const ist = ratio(werte(Rk, mode, fg), werte(Rk, mode, bg));
    if (ist >= soll) continue;
    const pf = primitiveVon(T, mode, fg), pb = primitiveVon(T, mode, bg);
    const surfaceVon = T.derived[mode][fg]?.surface;
    const gelieferteBeschriftung = surfaceVon && primitiveVon(T, mode, surfaceVon) in geliefert;
    const istGeliefert = (pf && pf in geliefert) || (pb && pb in geliefert) || Boolean(gelieferteBeschriftung);
    let vorschlag;
    try { vorschlag = aufKontrast(werte(Rk, mode, fg), [werte(Rk, mode, bg)], soll, mode === 'light' ? 'dunkler' : 'heller').hex; } catch { vorschlag = null; }
    befunde.push({ mode, fg, bg, ist, soll, vorschlag, geliefert: istGeliefert });
  }
  const gerechnet = befunde.filter((b) => !b.geliefert);
  if (gerechnet.length) {
    const e = new Error(`gerechnete Werte verfehlen ein Paar: ${gerechnet.map((b) => `${b.mode} ${b.fg}/${b.bg}`).join(', ')}`);
    e.code = 3; throw e;
  }

  return { theme, rollen, befunde, hinweise, ergebnis: Rk, css: css(theme, quelle, rollen, befunde, hinweise) };
}

function block(selector, eintraege) {
  return `${selector} {\n${Object.entries(eintraege).map(([n, v]) => `  --tl-${n}: ${v};`).join('\n')}\n}\n`;
}

function css(theme, quelle, rollen, befunde, hinweise) {
  const erster = { ...rollen.measure, ...rollen.light };
  let out = HERKUNFT(theme, quelle, befunde.length);
  out += block(`[data-tl-theme="${theme}"]`, erster);
  out += block(`[data-tl-theme="${theme}"][data-tl-mode="dark"]`, rollen.dark);
  if (befunde.length || hinweise.length) {
    out += '\n/* Report\n';
    for (const b of befunde) out += `   finding  ${b.mode}: ${b.fg} on ${b.bg} = ${b.ist.toFixed(2)}, required ${b.soll}${b.vorschlag ? `, suggestion ${b.vorschlag}` : ''}\n`;
    for (const h of hinweise) out += `   note     ${h}\n`;
    out += '*/\n';
  }
  return out;
}
```
